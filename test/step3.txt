# Step 3: traces, graph interaction, files (simulated Ultra).
#   del test\out\step3.ini & tinySA-App.exe --sim --ini test\out\step3.ini --script test\step3.txt --out test\out\step3
do connect sim:ultra
waitfor connected == 1
set startEdit 80
set stopEdit 120
set pointsCombo 101
do mark
click scanButton
waitfor sweeps.new >= 2
check peak.hz == 100000000
check peak.dbm ~ -30 2

# hover readout at the carrier
do mouse hover plot 100000000 -50
check hover contains 100.0 MHz
check hover contains Live -

# memories, normalisation, linear scale
do memory 1 Store live trace
check mem1 == 101
click normButton
do mark
waitfor sweeps.new >= 2
check level@100000000 ~ 0 3
check level@90000000 ~ 0 4
click normButton
set scaleCombo Linear (W)
check linear == 1
set scaleCombo Log (dBm)

# frequency strip: drag the middle third 10 MHz to the left = pan up 10 MHz
do mouse press xaxis 100000000
do mouse move xaxis 90000000
do mouse release xaxis 90000000
check start == 90000000
check stop == 130000000
# left third: the grabbed frequency follows the mouse, stop stays
do mouse press xaxis 92000000
do mouse move xaxis 97000000
do mouse release xaxis 97000000
check start == 85000000
check stop == 130000000
# wheel in the plot zooms about the cursor
do mouse wheelup plot 100000000 -50
check start == 87500000
check stop == 125000000
# level strip, top third: pull the top from -15 down to -25 dBm -> top rises 10 dB
do levels -110 -10
do mouse press yaxis 0 -15
do mouse move yaxis 0 -25
do mouse release yaxis 0 -25
check ymax ~ 0 0.5
check ymin ~ -110 0.5
# context menu and keys
do menu 110000000 -50 Center here
check start == 91250000
check stop == 128750000
key spectrum PgUp
check start == 97500000
check stop == 122500000
do menu 100000000 -50 Spline
do menu 100000000 -50 Show points
set calcCombo Max hold
set avgCombo 4
do mark
waitfor sweeps.new >= 3
shot step3_view.png

# files
do savecsv test/out/step3/trace.csv
do loadmem 2 test/out/step3/trace.csv
check mem2 == 101
do saveimage test/out/step3/image.png
key spectrum Escape
waitfor idle == 1
do recordfolder test/out/step3/rec
click actionRecord
do mark
click scanButton
waitfor sweeps.new >= 3
click scanButton
waitfor idle == 1
click actionRecord
check recording >= 3
check messages ==
quit
