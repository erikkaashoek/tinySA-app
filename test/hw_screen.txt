# Screen capture and live pushes on hardware (does not touch the tinySA screen):
#   tinySA-App.exe --port COM20 --ini test\out\hw.ini --script test\hw_screen.txt --out test\out\hwscreen
waitfor connected == 1 10000
click actionScreen
set pollCheck 1
waitfor screen.width == 480 8000
# RLE8 capture (tinySA4 firmware with __CAPTURE_RLE8__); older firmware sends "raw 307200"
check console.log contains RLE8
shotw screenWindow hw_capture.png
set remoteCheck 1
waitfor screen.updates >= 4 10000
shotw screenWindow hw_remote.png
set remoteCheck 0
waitfor idle == 1 8000
# refresh rle (firmware with __REFRESH_RLE__); older firmware answers usage, then refresh on
check sent contains refresh rle
check sent contains refresh off
check messages ==
dump console.log console.log
quit
